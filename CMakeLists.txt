cmake_minimum_required(VERSION 3.24)
project(MemDisk VERSION 0.1.6 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /utf-8)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
include_directories(${PROJECT_SOURCE_DIR}/include)
add_library(memdisk-common STATIC src/common/driver_client.cpp src/common/pipe_client.cpp
  src/common/restricted_token.cpp src/common/privilege.cpp src/common/detach.cpp)
target_link_libraries(memdisk-common PUBLIC advapi32)
target_include_directories(memdisk-common PUBLIC src/common)
add_executable(memdisk src/cli/main.cpp)
target_link_libraries(memdisk PRIVATE memdisk-common)
add_executable(memdisk-access-test src/tools/access_probe.cpp)
target_link_libraries(memdisk-access-test PRIVATE memdisk-common)
add_executable(memdisk-service src/service/main.cpp src/service/server.cpp
  src/service/operations.cpp src/service/volume.cpp src/service/permissions.cpp)
target_link_libraries(memdisk-service PRIVATE memdisk-common advapi32)
include(cmake/Driver.cmake)
include(cmake/Gui.cmake)
add_executable(memdisk-tests tests/protocol_tests.cpp tests/transport_tests.cpp tests/security_tests.cpp tests/detach_tests.cpp)
target_link_libraries(memdisk-tests PRIVATE memdisk-common)
enable_testing()
add_test(NAME protocol COMMAND memdisk-tests)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT memdisk-ui)
